> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build zhao_lio, a ROS2 LiDAR-inertial odometry that combines Super-LIO's closest-point voxel downsampling and OctVoxMap with Point-LIO-style point-wise 30-dim ESKF updates, with two switchable strategies (`pointwise`, `pointwise_prematch` with TBB parallel pre-association).

**Architecture:** A ROS-free C++17 core library (`zhao_lio_core`) holds all algorithms and is unit-tested on synthetic data. A ROS2 package (`zhao_lio_ros2`) wraps it with a live node and an offline bag replay executable that records TUM trajectories and per-stage timing CSV.

**Tech Stack:** C++17, Eigen 3.4, oneTBB 2021.5, ROS2 Humble (rclcpp, rosbag2_cpp), gtest via ament_cmake_gtest, optional livox_ros_driver2.

**Spec:** `zhao_lio/docs/specs/2026-10-29-zhao-lio-design.md`

```cpp
#pragma once

#include <cmath>

#include <Eigen/Core>

namespace zhao_lio {

inline Eigen::Matrix3d hat(const Eigen::Vector3d& v) {
  Eigen::Matrix3d m;
  m << 0, -v[2], v[1],
       v[2], 0, -v[0],
       -v[1], v[0], 0;
  return m;
}

inline Eigen::Matrix3d so3_exp(const Eigen::Vector3d& ang) {
  const double theta = ang.norm();
  if (theta < 1e-7) return Eigen::Matrix3d::Identity() + hat(ang);
  const Eigen::Matrix3d K = hat(ang / theta);
  return Eigen::Matrix3d::Identity() + std::sin(theta) * K + (1.0 - std::cos(theta)) * K * K;
}

// SO(3) Jacobian used by small_point_lio's covariance propagation (A_matrix).
inline Eigen::Matrix3d so3_A(const Eigen::Vector3d& v) {
  const double sq = v.squaredNorm();
  if (sq < 1e-22) return Eigen::Matrix3d::Identity();
  const double n = std::sqrt(sq);
  const Eigen::Matrix3d H = hat(v);
  return Eigen::Matrix3d::Identity() + (1.0 - std::cos(n)) / sq * H + (1.0 - std::sin(n) / n) / sq * H * H;
}

}  // namespace zhao_lio
```

```cpp
#include "zhao_lio/events.h"

namespace zhao_lio {

std::vector<Event> merge_events(const std::vector<LidarPoint>& points, const std::vector<ImuSample>& imus) {
  std::vector<Event> events;
  events.reserve(points.size() + imus.size());
  size_t i = 0, j = 0;
  while (i < points.size() || j < imus.size()) {
    const bool take_imu = j < imus.size() && (i >= points.size() || imus[j].time <= points[i].time);
    if (take_imu) {
      events.push_back({Event::kImu, imus[j].time, static_cast<uint32_t>(j)});
      ++j;
    } else {
      events.push_back({Event::kPoint, points[i].time, static_cast<uint32_t>(i)});
      ++i;
    }
  }
  return events;
}

}  // namespace zhao_lio
```

```cpp
#pragma once

#include <cstddef>
#include <deque>

#include "zhao_lio/types.h"

namespace zhao_lio {

// Pairs each LiDAR scan with all IMU samples up to its end time.
class MeasureSync {
 public:
  static constexpr std::size_t kMaxScans = 20;

  void push_imu(const ImuSample& imu);  // out-of-order samples are dropped
  void push_scan(ScanFrame scan);       // keeps at most kMaxScans waiting scans
  // Pops the oldest scan once the newest IMU sample reaches its end time.
  bool pop(Measure& out);
  std::size_t dropped_scans() const { return dropped_scans_; }

 private:
  std::deque<ImuSample> imus_;
  std::deque<ScanFrame> scans_;
  double last_imu_time_ = -1e300;
  std::size_t dropped_scans_ = 0;
};

}  // namespace zhao_lio
```

```cpp
#include "zhao_lio/sync.h"

namespace zhao_lio {

void MeasureSync::push_imu(const ImuSample& imu) {
  if (imu.time <= last_imu_time_) return;
  last_imu_time_ = imu.time;
  imus_.push_back(imu);
}

void MeasureSync::push_scan(ScanFrame scan) {
  scans_.push_back(std::move(scan));
  while (scans_.size() > kMaxScans) {
    scans_.pop_front();
    ++dropped_scans_;
  }
}

bool MeasureSync::pop(Measure& out) {
  if (scans_.empty() || imus_.empty()) return false;
  if (imus_.back().time < scans_.front().t_end) return false;
  out.scan = std::move(scans_.front());
  scans_.pop_front();
  out.imus.clear();
  while (!imus_.empty() && imus_.front().time <= out.scan.t_end) {
    out.imus.push_back(imus_.front());
    imus_.pop_front();
  }
  return true;
}

}  // namespace zhao_lio
```

```cpp
#include "zhao_lio/preprocess.h"

#include <algorithm>

namespace zhao_lio {

void preprocess(const ScanFrame& in, const Params& p, std::vector<LidarPoint>& out) {
  out.clear();
  out.reserve(in.points.size());
  const double min2 = p.blind_min * p.blind_min;
  const double max2 = p.blind_max * p.blind_max;
  const int step = std::max(1, p.point_filter_num);
  for (size_t i = 0; i < in.points.size(); i += step) {
    const LidarPoint& pt = in.points[i];
    if (!pt.xyz.allFinite()) continue;
    const double r2 = pt.xyz.squaredNorm();
    if (r2 < min2 || r2 > max2) continue;
    out.push_back(pt);
  }
}

void sort_by_time(std::vector<LidarPoint>& pts) {
  std::sort(pts.begin(), pts.end(), [](const LidarPoint& a, const LidarPoint& b) { return a.time < b.time; });
}

}  // namespace zhao_lio
```

```cpp
#pragma once

#include <algorithm>
#include <cstdint>
#include <vector>

#include <Eigen/Core>
#include <tbb/blocked_range.h>
#include <tbb/parallel_for.h>
#include <tbb/parallel_sort.h>
#include <tsl/robin_hood.h>

#include "zhao_lio/types.h"

namespace zhao_lio {

// Super-LIO VoxelGridClosest: per voxel keep the original point closest to the voxel center.
// Points are never averaged, so each kept point keeps its own timestamp.
class VoxelDownsample {
 public:
  explicit VoxelDownsample(float leaf = 0.5f) { set_leaf(leaf); }
  void set_leaf(float leaf) { leaf_ = leaf; inv_leaf_ = 1.0f / leaf; }
  float leaf() const { return leaf_; }

  // 21 bits per axis: valid for |coord| < 2^20 * leaf.
  inline void key_of(const Eigen::Vector3f& p, uint64_t& key, float& d2) const {
    const Eigen::Vector3i idx = (p * inv_leaf_).array().round().cast<int>();
    d2 = (p - leaf_ * idx.cast<float>()).squaredNorm();
    constexpr int64_t kOff = int64_t(1) << 20;
    constexpr uint64_t kMask = (uint64_t(1) << 21) - 1;
    key = ((static_cast<uint64_t>(idx[0] + kOff) & kMask) << 42) |
          ((static_cast<uint64_t>(idx[1] + kOff) & kMask) << 21) |
          (static_cast<uint64_t>(idx[2] + kOff) & kMask);
  }

  void filter_serial(const std::vector<LidarPoint>& in, std::vector<LidarPoint>& out) {
    map_.clear();
    map_.reserve(in.size());
    best_d2_.clear();
    out.clear();
    for (const auto& pt : in) {
      uint64_t key;
      float d2;
      key_of(pt.xyz, key, d2);
      auto it = map_.find(key);
      if (it == map_.end()) {
        map_.emplace(key, static_cast<uint32_t>(out.size()));
        out.push_back(pt);
        best_d2_.push_back(d2);
      } else if (d2 < best_d2_[it->second]) {
        out[it->second] = pt;
        best_d2_[it->second] = d2;
      }
    }
  }

  // Parallel key computation + parallel sort by (key, d2, index), then take the head of each run.
  // Ties resolve to the lowest index, same as filter_serial.
  void filter_parallel(const std::vector<LidarPoint>& in, std::vector<LidarPoint>& out) const {
    struct Entry { uint64_t key; float d2; uint32_t idx; };
    std::vector<Entry> e(in.size());
    tbb::parallel_for(tbb::blocked_range<size_t>(0, in.size(), 2048), [&](const tbb::blocked_range<size_t>& r) {
      for (size_t i = r.begin(); i < r.end(); ++i) {
        key_of(in[i].xyz, e[i].key, e[i].d2);
        e[i].idx = static_cast<uint32_t>(i);
      }
    });
    tbb::parallel_sort(e.begin(), e.end(), [](const Entry& a, const Entry& b) {
      if (a.key != b.key) return a.key < b.key;
      if (a.d2 != b.d2) return a.d2 < b.d2;
      return a.idx < b.idx;
    });
    out.clear();
    for (size_t i = 0; i < e.size(); ++i) {
      if (i == 0 || e[i].key != e[i - 1].key) out.push_back(in[e[i].idx]);
    }
  }

 private:
  float leaf_ = 0.5f;
  float inv_leaf_ = 2.0f;
  robin_hood::unordered_flat_map<uint64_t, uint32_t> map_;
  std::vector<float> best_d2_;
};

}  // namespace zhao_lio
```

```cpp
struct KnnHeap5 {
  using Point = Eigen::Vector3f;
  uint8_t count = 0;
  uint8_t worst = 0;
  float max_dist2 = 0.0f;
  std::array<float, 5> dist2{};
  std::array<Point, 5> points;

  void reset() { count = 0; worst = 0; max_dist2 = 0.0f; }

  inline void try_insert(float d2, const Point& pt) {
    if (count < 5) {
      dist2[count] = d2;
      points[count] = pt;
      if (count == 0 || d2 > max_dist2) { max_dist2 = d2; worst = count; }
      ++count;
      return;
    }
    if (d2 >= max_dist2) return;
    dist2[worst] = d2;
    points[worst] = pt;
    update_worst();
  }

 private:
  inline void update_worst() {
    const float d0 = dist2[0], d1 = dist2[1], d2 = dist2[2], d3 = dist2[3], d4 = dist2[4];
    const uint8_t i01 = d0 > d1 ? 0 : 1;
    const float m01 = d0 > d1 ? d0 : d1;
    const uint8_t i23 = d2 > d3 ? 2 : 3;
    const float m23 = d2 > d3 ? d2 : d3;
    const uint8_t i0123 = m01 > m23 ? i01 : i23;
    const float m0123 = m01 > m23 ? m01 : m23;
    worst = m0123 > d4 ? i0123 : 4;
    max_dist2 = m0123 > d4 ? m0123 : d4;
  }
};

// One voxel = 2x2x2 sub-voxels, each holding one running-mean representative point.
struct OctVox {
  static constexpr uint8_t kEmpty = 0;
  static constexpr uint8_t kMaxPoints = 20;
  static constexpr float kMergeDist2 = 0.1f * 0.1f;

  std::array<uint8_t, 8> counts{};
  std::array<Eigen::Vector3f, 8> points;

  OctVox(const Eigen::Vector3f& pt, uint8_t local_idx) { counts.fill(kEmpty); points[local_idx] = pt; counts[local_idx] = 1; }

  void add(const Eigen::Vector3f& pt, uint8_t local_idx) {
    uint8_t& c = counts[local_idx];
    Eigen::Vector3f& s = points[local_idx];
    if (c == kEmpty) { s = pt; c = 1; return; }
    if (c >= kMaxPoints) return;
    if ((pt - s).squaredNorm() > kMergeDist2) return;
    s = (s * c + pt) / static_cast<float>(c + 1);
    ++c;
  }

  bool get(uint8_t local_idx, Eigen::Vector3f& pt) const {
    if (counts[local_idx] == kEmpty) return false;
    pt = points[local_idx];
    return true;
  }
};
```

```cpp
  // 5 nearest stored points using Super-LIO's precomputed sub-voxel search order with early stop.
  void get_top_k(const Point& q, KnnHeap5& top) const {
    top.reset();
    const Key fine = (q * sub_inv_resolution_).array().floor().template cast<int>();
    const Key key(fine[0] >> 1, fine[1] >> 1, fine[2] >> 1);
    const int dx = fine[0] & 1, dy = fine[1] & 1, dz = fine[2] & 1;
    const uint8_t local_idx = static_cast<uint8_t>((dz << 2) | (dy << 1) | dx);
    const Key mirror(1 - (dx << 1), 1 - (dy << 1), 1 - (dz << 1));

    constexpr int kPre = 8;
    const OctVox* pre[kPre] = {nullptr};
    for (int i = 0; i < kPre; ++i) {
      auto it = grids_.find(key + mirror.cwiseProduct(HKNN_neighbor_voxel[i]));
      if (it != grids_.end()) pre[i] = &it->second->second;
    }

    Point sub;
    for (int g = 0; g < group_max_; ++g) {
      const uint8_t* it = group_ptrs_[g];
      const uint8_t* end = group_ptrs_[g + 1];
      while (it < end) {
        const uint8_t neighbor = *it++;
        uint8_t n = *it++;
        const OctVox* vox = nullptr;
        if (neighbor < kPre) {
          vox = pre[neighbor];
        } else {
          auto f = grids_.find(key + mirror.cwiseProduct(HKNN_neighbor_voxel[neighbor]));
          if (f != grids_.end()) vox = &f->second->second;
        }
        if (!vox) { it += n; continue; }
        while (n--) {
          const uint8_t li = static_cast<uint8_t>((*it++) ^ local_idx);
          if (vox->get(li, sub)) top.try_insert((sub - q).squaredNorm(), sub);
        }
      }
      if (top.count == 5 && top.max_dist2 < min_dis2_[g]) break;
    }
  }

 private:
  struct KeyHash {
    std::size_t operator()(const Key& v) const {
      std::size_t h = static_cast<std::size_t>(v[0]);
      h ^= v[1] * 0x9e3779b9 + (h << 6) + (h >> 2);
      h ^= v[2] * 0x85ebca6b + (h << 6) + (h >> 2);
      return h;
    }
  };
  using DataList = std::list<std::pair<Key, OctVox>>;

  inline void split_key(const Point& pt, Key& key, uint8_t& local_idx) const {
    const Key fine = (pt * sub_inv_resolution_).array().floor().template cast<int>();

    local_idx = static_cast<uint8_t>(((fine[2] & 1) << 2) | ((fine[1] & 1) << 1) | (fine[0] & 1));

  std::vector<const uint8_t*> group_ptrs_;
  std::array<float, 6> min_dis2_{};
  int group_max_ = 0;
```

```cpp
#include "zhao_lio/plane.h"

#include <cmath>

#include <Eigen/QR>

namespace zhao_lio {

Plane fit_plane(const KnnHeap5& nn, double thr) {
  Plane pl;
  const int n = nn.count;
  if (n < 4) return pl;
  Eigen::Matrix<double, Eigen::Dynamic, 3, 0, 5, 3> A(n, 3);
  Eigen::Matrix<double, Eigen::Dynamic, 1, 0, 5, 1> b(n);
  for (int j = 0; j < n; ++j) {
    A.row(j) = nn.points[j].cast<double>().transpose();
    b(j) = -1.0;
  }
  const Eigen::Vector3d nv = A.colPivHouseholderQr().solve(b);
  const double norm = nv.norm();
  if (norm < 1e-6) return pl;
  pl.n = nv / norm;
  pl.d = 1.0 / norm;
  for (int j = 0; j < n; ++j) {
    if (std::abs(pl.n.dot(nn.points[j].cast<double>()) + pl.d) > thr) return pl;
  }
  pl.valid = true;
  return pl;
}

Plane match_plane(const OctVoxMap& map, const Eigen::Vector3d& p_world, double thr) {
  KnnHeap5 nn;
  map.get_top_k(p_world.cast<float>(), nn);
  return fit_plane(nn, thr);
}
```

```cpp
void State30::plus(const Vec& dx) {

  rot = rot * so3_exp(dx.segment<3>(kRot));
  R_LI = R_LI * so3_exp(dx.segment<3>(kRLI));
  t_LI += dx.segment<3>(kTLI);
  vel += dx.segment<3>(kVel);
  omg += dx.segment<3>(kOmg);
  acc += dx.segment<3>(kAcc);
  grav += dx.segment<3>(kGrav);
  bg += dx.segment<3>(kBg);
  ba += dx.segment<3>(kBa);
}

Eskf30::Eskf30(const EskfOptions& opt) : opt_(opt) {
  Q_.setZero();
  Q_.block<3, 3>(S::kVel, S::kVel).diagonal().setConstant(opt.vel_cov);
  Q_.block<3, 3>(S::kOmg, S::kOmg).diagonal().setConstant(opt.omg_cov);
  Q_.block<3, 3>(S::kAcc, S::kAcc).diagonal().setConstant(opt.acc_cov);
  Q_.block<3, 3>(S::kBg, S::kBg).diagonal().setConstant(opt.bg_cov);
  Q_.block<3, 3>(S::kBa, S::kBa).diagonal().setConstant(opt.ba_cov);
  reset_covariance();
}

// Initial values from small_point_lio Estimator::reset(). With a fixed extrinsic its rows and
// columns are zero, so no update can move it.
void Eskf30::reset_covariance() {
  P = Mat::Identity() * 0.01;
  P.block<3, 3>(S::kGrav, S::kGrav).diagonal().setConstant(1e-4);
  P.block<3, 3>(S::kBg, S::kBg).diagonal().setConstant(1e-3);
  P.block<3, 3>(S::kBa, S::kBa).diagonal().setConstant(1e-3);
  if (!opt_.extrinsic_est) {
    P.middleRows<6>(S::kRLI).setZero();
    P.middleCols<6>(S::kRLI).setZero();
  }
}

void Eskf30::predict_state(double t) {
  const double dt = t - t_state_;
  if (dt <= 0.0) return;
  t_state_ = t;
  x.pos += x.vel * dt;
  x.rot = x.rot * so3_exp(x.omg * dt);
  x.vel += (x.rot * x.acc + x.grav) * dt;
}

void Eskf30::predict_cov(double t) {
  const double dt = t - t_cov_;
  if (dt <= 0.0) return;
  t_cov_ = t;
  const Eigen::Vector3d seg = -x.omg * dt;
  Mat F = Mat::Identity();
  F.block<3, 3>(S::kPos, S::kVel).diagonal().setConstant(dt);
  F.block<3, 3>(S::kRot, S::kRot) = so3_exp(seg);
  F.block<3, 3>(S::kRot, S::kOmg) = so3_A(seg) * dt;
  F.block<3, 3>(S::kVel, S::kRot) = -x.rot * hat(x.acc) * dt;  // small_point_lio omits dt here
  F.block<3, 3>(S::kVel, S::kAcc) = x.rot * dt;
  F.block<3, 3>(S::kVel, S::kGrav).diagonal().setConstant(dt);
  P = F * P * F.transpose() + Q_ * (dt * dt);
}
```

```cpp
// z = [gyro - ω - bg ; acc - a - ba]. Row k of H has +1 at (ω|a)_k and (bg|ba)_k.
// Saturated axes are skipped; the accelerometer uses flag k+3 (small_point_lio reads flag k).
bool Eskf30::update_imu(const Eigen::Vector3d& gyro, const Eigen::Vector3d& acc) {
  bool satu[6] = {false, false, false, false, false, false};
  Eigen::Matrix<double, 6, 1> z;
  z.head<3>() = gyro - x.omg - x.bg;
  z.tail<3>() = acc - x.acc - x.ba;
  if (opt_.check_satu) {
    for (int i = 0; i < 3; ++i) {
      if (std::abs(gyro[i]) >= opt_.satu_gyro) { satu[i] = true; z[i] = 0.0; }
      if (std::abs(acc[i]) >= opt_.satu_acc) { satu[i + 3] = true; z[i + 3] = 0.0; }
    }
  }
  auto col_a = [](int k) { return k < 3 ? S::kOmg + k : S::kAcc + k - 3; };
  auto col_b = [](int k) { return k < 3 ? S::kBg + k : S::kBa + k - 3; };

  Eigen::Matrix<double, S::kDim, 6> PHt = Eigen::Matrix<double, S::kDim, 6>::Zero();
  Eigen::Matrix<double, 6, S::kDim> HP = Eigen::Matrix<double, 6, S::kDim>::Zero();
  for (int k = 0; k < 6; ++k) {
    if (satu[k]) continue;
    PHt.col(k) = P.col(col_a(k)) + P.col(col_b(k));
    HP.row(k) = P.row(col_a(k)) + P.row(col_b(k));
  }
  Eigen::Matrix<double, 6, 6> Smat = Eigen::Matrix<double, 6, 6>::Zero();
  for (int k = 0; k < 6; ++k) {
    if (!satu[k]) Smat.col(k) = HP.col(col_a(k)) + HP.col(col_b(k));
    Smat(k, k) += k < 3 ? opt_.imu_meas_omg_cov : opt_.imu_meas_acc_cov;
  }
  Eigen::LDLT<Eigen::Matrix<double, 6, 6>> ldlt(Smat);
  if (ldlt.info() != Eigen::Success) return false;
  const Eigen::Matrix<double, S::kDim, 6> K = PHt * ldlt.solve(Eigen::Matrix<double, 6, 6>::Identity());
  x.plus(K * z);
  P -= K * HP;
  return true;
}

// r = nᵀ(R(R_LI p + t_LI) + pos) + d, H = [nᵀ, (p_imu × Rᵀn)ᵀ, (p × R_LIᵀRᵀn)ᵀ, (Rᵀn)ᵀ], rank-1 update.
bool Eskf30::update_point(const Eigen::Vector3d& n, double d, const Eigen::Vector3d& p_lidar) {
  const Eigen::Vector3d p_imu = x.R_LI * p_lidar + x.t_LI;
  const double r = n.dot(x.rot * p_imu + x.pos) + d;
  const Eigen::Vector3d C = x.rot.transpose() * n;
  Eigen::Matrix<double, 1, 12> H;
  if (opt_.extrinsic_est) {
    const Eigen::Vector3d B = p_lidar.cross(x.R_LI.transpose() * C);
    H << n.transpose(), p_imu.cross(C).transpose(), B.transpose(), C.transpose();
  } else {
    H << n.transpose(), p_imu.cross(C).transpose(), Eigen::Matrix<double, 1, 6>::Zero();
  }
  const Vec PHt = P.leftCols<12>() * H.transpose();
  const double s = H.dot(PHt.head<12>()) + opt_.laser_point_cov;
  if (s < 1e-12) return false;
  const Vec K = PHt / s;
  x.plus(K * (-r));
  P -= K * (H * P.topRows<12>());
  return true;
}
```

```cpp
// Box room [-5,5]x[-5,5]x[-1.5,2.5]; LiDAR frame == IMU frame. Static until t_static, then
// x, y and yaw follow a·(1 - cos(w·τ)), τ = t - t_static.
struct Trajectory {
  double t_static = 1.5, ax = 1.0, ay = 0.6, ayaw = 0.5, w = 0.6, g = 9.81;

  double tau(double t) const { return std::max(0.0, t - t_static); }
  double c(double t) const { return std::cos(w * tau(t)); }
  double s(double t) const { return t > t_static ? std::sin(w * tau(t)) : 0.0; }
  Eigen::Vector3d pos(double t) const { return {ax * (1 - c(t)), ay * (1 - c(t)), 0.0}; }
  Eigen::Vector3d acc_world(double t) const {
    if (t <= t_static) return Eigen::Vector3d::Zero();
    return {ax * w * w * c(t), ay * w * w * c(t), 0.0};
  }
  double yaw(double t) const { return ayaw * (1 - c(t)); }
  double yaw_rate(double t) const { return ayaw * w * s(t); }
  Eigen::Matrix3d rot(double t) const { return Eigen::AngleAxisd(yaw(t), Eigen::Vector3d::UnitZ()).toRotationMatrix(); }
};

inline ImuSample make_imu(const Trajectory& tr, double t) {
  ImuSample s;
  s.time = t;
  s.gyro = Eigen::Vector3d(0, 0, tr.yaw_rate(t));
  s.acc = tr.rot(t).transpose() * (tr.acc_world(t) + Eigen::Vector3d(0, 0, tr.g));
  return s;
}

inline double ray_box(const Eigen::Vector3d& o, const Eigen::Vector3d& d) {
```

```cpp
// One scan of n points over (t0, t0 + dt]: elevation −30°..60°, golden-angle azimuth, 5 mm range noise.
inline ScanFrame make_scan(const Trajectory& tr, double t0, double dt, int n, std::mt19937& rng) {
  std::normal_distribution<double> noise(0.0, 0.005);
  const double golden = M_PI * (3.0 - std::sqrt(5.0));
  ScanFrame f;
  f.t_begin = t0;
  f.t_end = t0 + dt;
  f.points.reserve(n);
  for (int k = 0; k < n; ++k) {
    const double t = t0 + dt * (k + 1) / n;
    const double el = (-30.0 + 90.0 * std::fmod(k * 0.618034, 1.0)) * M_PI / 180.0;
    const double az = golden * k;
    const Eigen::Vector3d d_body(std::cos(el) * std::cos(az), std::cos(el) * std::sin(az), std::sin(el));
    const double range = ray_box(tr.pos(t), tr.rot(t) * d_body) + noise(rng);
    LidarPoint p;
    p.xyz = (d_body * range).cast<float>();
    p.time = t;
    f.points.push_back(p);
  }
  return f;
```

```cpp
#include "zhao_lio/updater/pointwise_prematch.h"

#include <cmath>

#include <tbb/blocked_range.h>
#include <tbb/parallel_for.h>

namespace zhao_lio {

void PointwisePrematchStrategy::process(const StrategyInput& in, Eskf30& kf, const OctVoxMap& map,
                                        StrategyOutput& out) {
  reset_output(in, out);
  const size_t n = in.points.size();
  pred_world_.resize(n);
  planes_.assign(n, Plane());

  const auto t0 = std::chrono::steady_clock::now();
  {  // Phase 1: nominal prediction on a copy; IMU samples overwrite ω and a with bias-corrected readings.
    Eskf30 pred = kf;
    for (const Event& ev : in.events) {
      if (ev.type == Event::kImu) {
        const ImuSample& imu = in.imus[ev.index];
        pred.predict_state(imu.time);
        pred.x.omg = imu.gyro - pred.x.bg;
        pred.x.acc = imu.acc - pred.x.ba;
      } else {
        pred.predict_state(ev.time);
        pred_world_[ev.index] = pred.x.lidar_to_world(in.points[ev.index].xyz.cast<double>());
      }
    }
  }
  // Phase 2: parallel association.
  tbb::parallel_for(tbb::blocked_range<size_t>(0, n, 64), [&](const tbb::blocked_range<size_t>& r) {
    for (size_t i = r.begin(); i < r.end(); ++i) planes_[i] = match_plane(map, pred_world_[i], params_.plane_thr);
  });
  out.stats.t_prematch_ms = ms_since(t0);

  // Phase 3: serial point-wise update.
  const auto t1 = std::chrono::steady_clock::now();
  for (const Event& ev : in.events) {
    if (ev.type == Event::kImu) {
      imu_step(kf, in.imus[ev.index]);
      continue;
    }
    const Eigen::Vector3d p = in.points[ev.index].xyz.cast<double>();
    if (ev.time >= kf.time()) {
      kf.predict_state(ev.time);
      Plane pl = planes_[ev.index];
      if (pl.valid) {
        double r = kf.point_residual(pl.n, pl.d, p);
        if (std::abs(r) > params_.rematch_thr) {
          ++out.stats.rematched;
          pl = match_plane(map, kf.x.lidar_to_world(p), params_.plane_thr);
          if (pl.valid) r = kf.point_residual(pl.n, pl.d, p);
        }
        if (pl.valid && pass_gate(p, r) && kf.update_point(pl.n, pl.d, p)) {
          ++out.stats.points_used;
          record(ev.time, kf, out);
        }
      }
    }
    out.points_world[ev.index] = kf.x.lidar_to_world(p).cast<float>();
  }
  out.stats.t_update_ms = ms_since(t1);
}
```

```cpp
void ZhaoLio::init_from_imu() {
```

```cpp
bool ZhaoLio::process(const Measure& m, ScanResult& out) {
  if (status_ == Status::kWaitImuInit) {
    init_imus_.insert(init_imus_.end(), m.imus.begin(), m.imus.end());
    if (!init_imus_.empty() && static_cast<int>(init_imus_.size()) >= p_.imu_init_num) {
      init_from_imu();
      status_ = Status::kWaitMapInit;
    }
    return false;
  }

  if (status_ == Status::kWaitMapInit) {
    for (const auto& imu : m.imus) imu_step(kf_, imu);
    FrameTiming unused;
    filter_and_downsample(m.scan, unused);
    std::vector<Eigen::Vector3f> world(down_.size());
    for (size_t i = 0; i < down_.size(); ++i) world[i] = kf_.x.lidar_to_world(down_[i].xyz.cast<double>()).cast<float>();
    map_->insert(world);
    if (++map_frames_ >= p_.init_map_frames) status_ = Status::kProcess;
    return false;
  }

  const auto t_total = Clock::now();
  out.timing = FrameTiming();
  out.num_raw = m.scan.points.size();
  filter_and_downsample(m.scan, out.timing);
  out.num_downsampled = down_.size();

  const std::vector<Event> events = merge_events(down_, m.imus);
  strategy_->process({down_, m.imus, events}, kf_, *map_, strat_out_);
  kf_.predict_state(m.scan.t_end);

  if (!kf_.x.pos.allFinite() || !kf_.x.rot.allFinite() || !kf_.x.vel.allFinite()) {
    std::fprintf(stderr, "[zhao_lio] state became non-finite, resetting\n");
    reset();
    return false;
  }

  const auto t_map = Clock::now();
  map_->insert(strat_out_.points_world);
  out.timing.map_ms = ms_since(t_map);
  out.timing.prematch_ms = strat_out_.stats.t_prematch_ms;
  out.timing.update_ms = strat_out_.stats.t_update_ms;
  out.timing.total_ms = ms_since(t_total);

  out.time = m.scan.t_end;
  out.state = kf_.x;
  out.stats = strat_out_.stats;
  out.points_world.swap(strat_out_.points_world);
  out.state_trace.swap(strat_out_.state_trace);
  return true;
}
```

```cpp
inline zhao_lio::ImuSample imu_from_msg(const sensor_msgs::msg::Imu& m, double acc_scale) {
  zhao_lio::ImuSample s;
  s.time = rclcpp::Time(m.header.stamp).seconds();
  s.gyro << m.angular_velocity.x, m.angular_velocity.y, m.angular_velocity.z;
  s.acc << m.linear_acceleration.x, m.linear_acceleration.y, m.linear_acceleration.z;
  s.acc *= acc_scale;
  return s;
}

inline void finish_scan(zhao_lio::ScanFrame& f, double header_time) {
  f.t_begin = header_time;
  f.t_end = header_time;
  for (const auto& p : f.points) f.t_end = std::max(f.t_end, p.time);
}

#ifdef HAVE_LIVOX_DRIVER
// Keeps first/strongest returns like Point-LIO's Livox handler ((tag & 0x30) is 0x00 or 0x10).
inline zhao_lio::ScanFrame scan_from_livox(const livox_ros_driver2::msg::CustomMsg& m) {
  zhao_lio::ScanFrame f;
  f.points.reserve(m.points.size());
  const double base = static_cast<double>(m.timebase) * 1e-9;
  for (const auto& pt : m.points) {
    const int ret = pt.tag & 0x30;
    if (ret != 0x00 && ret != 0x10) continue;
    zhao_lio::LidarPoint p;
    p.xyz = Eigen::Vector3f(pt.x, pt.y, pt.z);
    p.time = base + static_cast<double>(pt.offset_time) * 1e-9;
    f.points.push_back(p);
  }
  finish_scan(f, base);
  return f;
}
#endif
```

```cpp
// livox_ros_driver2 PointCloud2 (xfer_format 0): float32 x/y/z and a float64 per-point
// "timestamp". Values above 1e12 are treated as nanoseconds, otherwise seconds.
// Returns false when the cloud has no "timestamp" field.
inline bool scan_from_pointcloud2(const sensor_msgs::msg::PointCloud2& m, zhao_lio::ScanFrame& f) {
  const bool has_ts = std::any_of(m.fields.begin(), m.fields.end(), [](const auto& fd) { return fd.name == "timestamp"; });
  if (!has_ts) return false;
  f.points.clear();
  f.points.reserve(static_cast<size_t>(m.width) * m.height);
  sensor_msgs::PointCloud2ConstIterator<float> x(m, "x"), y(m, "y"), z(m, "z");
  sensor_msgs::PointCloud2ConstIterator<double> ts(m, "timestamp");
  for (; x != x.end(); ++x, ++y, ++z, ++ts) {
    zhao_lio::LidarPoint p;
    p.xyz = Eigen::Vector3f(*x, *y, *z);
    p.time = *ts > 1e12 ? *ts * 1e-9 : *ts;
    f.points.push_back(p);
  }
  finish_scan(f, rclcpp::Time(m.header.stamp).seconds());
  return true;
}
```

```cpp
 private:
  void spin_measures() {
    zhao_lio::Measure m;
    while (sync_.pop(m)) {
      if (!lio_->process(m, res_)) continue;
      tum_.write(res_.time, res_.state);
      csv_.write(res_);
      publish(res_);
      if (++frames_ % 100 == 0) {
        RCLCPP_INFO(get_logger(), "frame %d: total %.2f ms (pre %.2f ds %.2f match %.2f upd %.2f map %.2f) used %d/%zu rematch %d",
                    frames_, res_.timing.total_ms, res_.timing.preprocess_ms, res_.timing.downsample_ms,
                    res_.timing.prematch_ms, res_.timing.update_ms, res_.timing.map_ms, res_.stats.points_used,
                    res_.num_downsampled, res_.stats.rematched);
      }
    }
  }

  static void fill_pose(const zhao_lio::State30& s, geometry_msgs::msg::Pose& pose) {
    const Eigen::Quaterniond q(s.rot);
    pose.position.x = s.pos.x(); pose.position.y = s.pos.y(); pose.position.z = s.pos.z();
    pose.orientation.x = q.x(); pose.orientation.y = q.y(); pose.orientation.z = q.z(); pose.orientation.w = q.w();
  }

  void publish_odom(double t, const zhao_lio::State30& s) {
    nav_msgs::msg::Odometry od;
    od.header.stamp = rclcpp::Time(static_cast<int64_t>(t * 1e9));
    od.header.frame_id = cfg_.odom_frame;
    od.child_frame_id = cfg_.body_frame;
    fill_pose(s, od.pose.pose);
    const Eigen::Vector3d v_body = s.rot.transpose() * s.vel;
    od.twist.twist.linear.x = v_body.x(); od.twist.twist.linear.y = v_body.y(); od.twist.twist.linear.z = v_body.z();
    od.twist.twist.angular.x = s.omg.x(); od.twist.twist.angular.y = s.omg.y(); od.twist.twist.angular.z = s.omg.z();
    pub_odom_->publish(od);
  }

  void publish(const zhao_lio::ScanResult& r) {
    for (const auto& ts : r.state_trace) publish_odom(ts.first, ts.second);  // only when publish_pointwise_odom
    publish_odom(r.time, r.state);
    const rclcpp::Time stamp(static_cast<int64_t>(r.time * 1e9));

    geometry_msgs::msg::TransformStamped tf;
    tf.header.stamp = stamp;
    tf.header.frame_id = cfg_.odom_frame;
    tf.child_frame_id = cfg_.body_frame;
    tf.transform.translation.x = r.state.pos.x(); tf.transform.translation.y = r.state.pos.y(); tf.transform.translation.z = r.state.pos.z();
    const Eigen::Quaterniond q(r.state.rot);
    tf.transform.rotation.x = q.x(); tf.transform.rotation.y = q.y(); tf.transform.rotation.z = q.z(); tf.transform.rotation.w = q.w();
    tf_->sendTransform(tf);

    geometry_msgs::msg::PoseStamped ps;
    ps.header = tf.header;
    fill_pose(r.state, ps.pose);
    path_.header.stamp = stamp;
    path_.poses.push_back(ps);
    pub_path_->publish(path_);

    sensor_msgs::msg::PointCloud2 cloud;
    cloud.header = tf.header;
    sensor_msgs::PointCloud2Modifier mod(cloud);
    mod.setPointCloud2FieldsByString(1, "xyz");
    mod.resize(r.points_world.size());
    sensor_msgs::PointCloud2Iterator<float> x(cloud, "x"), y(cloud, "y"), z(cloud, "z");
    for (const auto& p : r.points_world) { *x = p.x(); *y = p.y(); *z = p.z(); ++x; ++y; ++z; }
    pub_cloud_->publish(cloud);
```

```cpp
#include <cstdio>

#include <rclcpp/rclcpp.hpp>
#include <rclcpp/serialization.hpp>
#include <rosbag2_cpp/reader.hpp>

#include "zhao_lio/zhao_lio.h"
#include "zhao_lio/io.h"
#include "zhao_lio/sync.h"
#include "zhao_lio_ros2/config.h"
#include "zhao_lio_ros2/msg_conversions.h"

// Reads a bag directly and feeds zhao_lio in timestamp order, without ROS transport, so per-frame
// timing does not depend on playback rate.
int main(int argc, char** argv) {
  rclcpp::init(argc, argv);
  auto node = std::make_shared<rclcpp::Node>("zhao_lio");
  const zhao_lio_ros2::NodeConfig cfg = zhao_lio_ros2::load_config(*node);
  const std::string bag = node->declare_parameter<std::string>("bag", "");
  if (bag.empty()) { std::fprintf(stderr, "set -p bag:=<path>\n"); return 1; }

  rosbag2_cpp::Reader reader;
  reader.open(bag);
  rclcpp::Serialization<sensor_msgs::msg::Imu> imu_ser;
  rclcpp::Serialization<sensor_msgs::msg::PointCloud2> pc2_ser;
#ifdef HAVE_LIVOX_DRIVER
  rclcpp::Serialization<livox_ros_driver2::msg::CustomMsg> livox_ser;
#else
  if (cfg.lidar_type == "livox_custom") { std::fprintf(stderr, "built without livox_ros_driver2\n"); return 1; }
#endif

  zhao_lio::ZhaoLio lio(cfg.params);
  zhao_lio::MeasureSync sync;
  zhao_lio::TumWriter tum;
  zhao_lio::TimingCsv csv;
  if (!cfg.traj_path.empty()) tum.open(cfg.traj_path);
  if (!cfg.timing_csv.empty()) csv.open(cfg.timing_csv);

  zhao_lio::FrameTiming sum;
  long frames = 0, used = 0, rematched = 0, down = 0;
  zhao_lio::Measure m;
  zhao_lio::ScanResult r;
  while (reader.has_next()) {
    auto msg = reader.read_next();
    rclcpp::SerializedMessage ser(*msg->serialized_data);
    if (msg->topic_name == cfg.imu_topic) {
      sensor_msgs::msg::Imu imu;
      imu_ser.deserialize_message(&ser, &imu);
      sync.push_imu(zhao_lio_ros2::imu_from_msg(imu, cfg.acc_scale));
    } else if (msg->topic_name == cfg.lidar_topic) {
      zhao_lio::ScanFrame f;
      if (cfg.lidar_type == "pointcloud2") {
        sensor_msgs::msg::PointCloud2 pc;
        pc2_ser.deserialize_message(&ser, &pc);
        if (!zhao_lio_ros2::scan_from_pointcloud2(pc, f)) { std::fprintf(stderr, "no 'timestamp' field\n"); return 1; }
      } else {
#ifdef HAVE_LIVOX_DRIVER
        livox_ros_driver2::msg::CustomMsg lm;
        livox_ser.deserialize_message(&ser, &lm);
        f = zhao_lio_ros2::scan_from_livox(lm);
#endif
      }
      sync.push_scan(std::move(f));
    } else {
      continue;
    }
```
